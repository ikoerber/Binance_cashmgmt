       async def _post_reconnect_reconciliation(self, user_id: str, start_time_iso: Optional[str]):
           # Debounce: skip if last reconciliation was < 60 seconds ago
           now = datetime.now(timezone.utc)
           if self._last_reconciliation_at and (now - self._last_reconciliation_at).total_seconds() < 60:
               logger.debug("Post-reconnect reconciliation skipped (debounce): user=%s", user_id)
               return
           try:
               self._last_reconciliation_at = now
               await asyncio.to_thread(self._sync_reconcile_fills, user_id, start_time_iso)
               logger.info("Post-reconnect reconciliation completed: user=%s", user_id)
           except Exception:

           from app.services.reconciliation_service import ReconciliationService
           from app.symbol_registry import KNOWN_PAIRS

           api_key = os.getenv("BINANCE_API_KEY")
           api_secret = os.getenv("BINANCE_API_SECRET")
           testnet = os.getenv("BINANCE_TESTNET", "").lower() == "true"

           binance_service = BinanceService(api_key, api_secret, testnet=testnet)
           recon_service = ReconciliationService(binance_service)
